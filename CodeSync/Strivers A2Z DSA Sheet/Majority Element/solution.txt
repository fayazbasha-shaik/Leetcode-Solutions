class Solution {
    public int majorityElement(int[] nums) {
        int max=Integer.MIN_VALUE;
        for(int i =0;i<nums.length;i++){
            int counter=0;
            for(int j=0;j<nums.length;j++){
                if(j==i){
                    counter++;
                }
        int value=0;
            }
            if(max<counter){
                max=counter;
                int value=j;
            }
        }
        return value;
        
    }

}
