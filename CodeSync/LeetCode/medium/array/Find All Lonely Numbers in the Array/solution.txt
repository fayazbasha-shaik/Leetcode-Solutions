int count = 0;
            boolean adjacent = false;

            for (int j = 0; j < nums.length; j++) {

                if (nums[i] == nums[j]) {
                    count++;
                }

                if (nums[i] - 1 == nums[j] ||
                    nums[i] + 1 == nums[j]) {
                    adjacent = true;
                }
            }

            if (count == 1 && !adjacent) {
                ans.add(nums[i]);
            }
        }

        return ans;
    }
