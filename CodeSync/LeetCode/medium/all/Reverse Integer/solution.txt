        int sign=1;
        if(x<0){
            sign=-1;
        }
            x=-x;
            int lastDigit=x%10;
            
            rev=rev*10+lastDigit;
            x=x/10;
            
        
        if(rev> Integer.MIN_VALUE && rev <Integer.MIN_value){
            return 0;
        }
return rev*sign;
