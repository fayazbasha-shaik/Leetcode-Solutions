class Solution {
    public int findMaxConsecutiveOnes(int[] nums) {
        int max=0;
        int start=0;
        int end=nums.length-1;
        int count=0;
       while(start<=end){
       if(nums[start]==1){
        count++;
        max=Math.max(max,count);
       }
       else{
        count=0;
       }
       start++;
