class Solution {
    public boolean isMonotonic(int[] nums) {
        int lp=0;
        int rp=nums.length-1;
        while(lp<rp){
            if(nums(lp)!<=nums && nums(lp)!>=nums(rp)){
        }
        
                return false 
            }
    }
        return true;
}
