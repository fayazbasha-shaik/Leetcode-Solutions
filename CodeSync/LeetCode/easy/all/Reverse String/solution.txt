class Solution {
    public void reverseString(char[] s) {
        String arr[]=new String[char.length];
        for(int i=0;i<char.length/2;i++){
            char temp=char[i];
            arr[i]=char[char.length-1-i];
            arr[arr.length-1-i]=temp;

        }
    }
}
