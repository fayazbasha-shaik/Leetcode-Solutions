class Solution {
    public int findMaxConsecutiveOnes(int[] nums) {
        int counter=Integer.MIN_VALUE;
        for(int i=0;i<nums.length;i++){
           int  count=0;
            for(int j=i+1;j<nums.length;j++){
                if(nums[i]==nums[j]){
                    count+=1;
             }
             if(counter<count){
                counter=count;
             }

            }

            
        }
        if(counter==Integer.MIN_VALUE){
            return 0;
        }
return counter;
        
    }
}
