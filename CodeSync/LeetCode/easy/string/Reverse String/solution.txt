class Solution {
    public void reverseString(char[] s) {
        String arr[]=new String[s.length];
        int i=0;
        int j=s.length;
        for(i=0;i<j/2;i++){
           s[i]=s[j-1-i];
           s[j-1-i]=s[i];

           j--;

        }
    }
}
